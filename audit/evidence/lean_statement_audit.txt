  (stp : SELLCert.StepTriple n m) (hk : σ.K stp.i = none) (j : Fin m) :
  SELLCert.syndrome H (SELLCert.applyStep H σ stp) j = SELLCert.syndrome H σ j + H j stp.i * stp.b

  (stp : SELLCert.StepTriple n m) (hk : σ.K stp.i = none) (j : Fin m) :
  SELLCert.degree H (SELLCert.applyStep H σ stp) j = SELLCert.degree H σ j - if H j stp.i ≠ 0 then 1 else 0
